from fastapi import FastAPI
from pyngrok import ngrok
import nest_asyncio
import uvicorn

# Allow nested event loops in Colab
nest_asyncio.apply()

# ----------------------------
# Create FastAPI app
# ----------------------------
app = FastAPI(title="Colab FastAPI")

@app.get("/")
async def root():
    return {
        "message": "FastAPI is running from Google Colab!"
    }

@app.get("/hello/{name}")
async def hello(name: str):
    return {
        "hello": name
    }

@app.post("/sum")
async def add_numbers(a: int, b: int):
    return {
        "sum": a + b
    }

# ----------------------------
# NGROK
# ----------------------------

# !pip -q install fastapi uvicorn pyngrok nest_asyncio

# Replace with your ngrok auth token
NGROK_AUTH_TOKEN = "YOUR_NGROK_AUTH_TOKEN"

ngrok.set_auth_token(NGROK_AUTH_TOKEN)

# Close any previous tunnels
ngrok.kill()

# Create public URL
public_url = ngrok.connect(8000)

print("=" * 60)
print("Public URL:", public_url.public_url)
print("Swagger Docs:", public_url.public_url + "/docs")
print("=" * 60)

# ----------------------------
# Start FastAPI
# ----------------------------
config = uvicorn.Config(
    app,
    host="0.0.0.0",
    port=8000,
    log_level="info"
)

server = uvicorn.Server(config)

await server.serve()
